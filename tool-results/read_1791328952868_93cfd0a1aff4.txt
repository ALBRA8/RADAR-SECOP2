    23→    23→export interface JsonRpcMessage {
    24→    24→  jsonrpc?: string
    25→    25→  id?: string | number | null
    26→    26→  method?: string
    27→    27→  params?: Record<string, unknown>
    28→    28→  result?: unknown
    29→    29→  error?: { code: number; message: string; data?: unknown }
    30→    30→}
    31→    31→
    32→    32→const ERR = {
    33→    33→  PARSE: { code: -32700, message: 'Error de parseo JSON' },
    34→    34→  REQUEST: { code: -32600, message: 'Request JSON-RPC inválido' },
    35→    35→  METHOD: { code: -32601, message: 'Método no encontrado' },
    36→    36→  PARAMS: { code: -32602, message: 'Parámetros inválidos' },
    37→    37→  INTERNAL: { code: -32603, message: 'Error interno' },
    38→    38→}
    39→    39→
    40→    40→function rpcResult(id: JsonRpcMessage['id'], result: Record<string, unknown>) {
    41→    41→  return { jsonrpc: '2.0', id, result }
    42→    42→}
    43→    43→
    44→    44→function rpcError(id: JsonRpcMessage['id'], err: { code: number; message: string }, data?: unknown) {
    45→    45→  return { jsonrpc: '2.0', id, error: { ...err, ...(data !== undefined ? { data } : {}) } }
    46→    46→}

   600→   600→export interface RpcOutcome {
   601→   601→  status: number
   602→   602→  body: JsonRpcMessage | JsonRpcMessage[] | null
   603→   603→  sessionId?: string
   604→   604→}
   605→   605→
   606→   606→export async function handleRpcMessage(msg: JsonRpcMessage): Promise<RpcOutcome> {
   607→   607→  if (!msg || typeof msg !== 'object' || msg.jsonrpc !== '2.0' || typeof msg.method !== 'string') {
   608→   608→    return { status: 400, body: rpcError(msg?.id ?? null, ERR.REQUEST) }
   609→   609→  }
   610→   610→
   611→   611→  const isNotification = msg.id === undefined || msg.id === null
   612→   612→
   613→   613→  switch (msg.method) {
   614→   614→    case 'initialize': {
   615→   615→      const requested = typeof msg.params?.protocolVersion === 'string' ? (msg.params.protocolVersion as string) : MCP_PROTOCOL_VERSION
   616→   616→      const version = requested.startsWith('2025-') || requested.startsWith('2024-') ? requested : MCP_PROTOCOL_VERSION
   617→   617→      return {
   618→   618→        status: 200,
   619→   619→        sessionId: crypto.randomUUID(),
   620→   620→        body: rpcResult(msg.id, {
   621→   621→          protocolVersion: version,
   622→   622→          capabilities: { tools: { listChanged: false } },
   623→   623→          serverInfo: MCP_SERVER_INFO,
   624→   624→          instructions:
   625→   625→            'Agente SECOP Radar: monitorea oportunidades de contratación pública colombiana (SECOP II), las analiza contra el perfil de la empresa, diseña proyectos-respuesta completos con Marco Lógico (árbol de problemas + matriz Fin/Propósito/Componentes/Actividades) y los refina conversacionalmente. También analiza pliegos/documentos y consulta la bandeja de canales de mensajería (Telegram/WhatsApp). Empieza con get_dashboard_stats o list_opportunities. Las tools de escritura respetan el flujo de aprobación humana.',
   626→   626→        }),
   627→   627→      }
   628→   628→    }
   629→   629→
   630→   630→    case 'notifications/initialized':
   631→   631→      return { status: 202, body: null }
   632→   632→
   633→   633→    case 'ping':
   634→   634→      return isNotification ? { status: 202, body: null } : { status: 200, body: rpcResult(msg.id, {}) }
   635→   635→
   636→   636→    case 'tools/list':
   637→   637→      return {
   638→   638→        status: 200,
   639→   639→        body: rpcResult(msg.id, {
   640→   640→          tools: Object.entries(MCP_TOOLS).map(([name, t]) => ({
   641→   641→            name,
   642→   642→            title: name,
   643→   643→            description: t.description + (t.write ? ' [escribe datos / exige aprobación humana previa]' : ''),
   644→   644→            inputSchema: t.inputSchema,
   645→   645→          })),
   646→   646→        }),
   647→   647→      }
   648→   648→  }
   649→   649→
   650→   650→  if (msg.method.startsWith('notifications/')) return { status: 202, body: null }
   651→   651→
   652→   652→  if (msg.method === 'tools/call') {
   653→   653→    const name = str(msg.params?.name)
   654→   654→    const tool = MCP_TOOLS[name]
   655→   655→    if (!tool) {
   656→   656→      return { status: 200, body: rpcError(msg.id, ERR.METHOD, `Tool desconocida: ${name}. Usa tools/list para ver las disponibles.`) }
   657→   657→    }
   658→   658→    const args = (msg.params?.arguments as Record<string, unknown>) || {}
   659→   659→    try {
   660→   660→      const result = await tool.handler(args)
   661→   661→      return {
   662→   662→        status: 200,
   663→   663→        body: rpcResult(msg.id, {
   664→   664→          content: [{ type: 'text', text: JSON.stringify(result, null, 2) }],
   665→   665→          structuredContent: result,
   666→   666→          isError: false,
   667→   667→        }),
   668→   668→      }
   669→   669→    } catch (err) {
   670→   670→      const message = err instanceof Error ? err.message : 'Error interno de la tool'
   671→   671→      return {
   672→   672→        status: 200,
   673→   673→        body: rpcResult(msg.id, { content: [{ type: 'text', text: `Error: ${message}` }], isError: true }),
   674→   674→      }
   675→   675→    }
   676→   676→  }
   677→   677→
   678→   678→  return { status: 200, body: rpcError(msg.id, ERR.METHOD, `Método no soportado: ${msg.method}`) }
   679→   679→}
