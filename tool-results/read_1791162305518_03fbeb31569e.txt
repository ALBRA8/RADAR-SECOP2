    22→export interface JsonRpcMessage {
    23→  jsonrpc?: string
    24→  id?: string | number | null
    25→  method?: string
    26→  params?: Record<string, unknown>
    27→  result?: unknown
    28→  error?: { code: number; message: string; data?: unknown }
    29→}
    30→
    31→const ERR = {
    32→  PARSE: { code: -32700, message: 'Error de parseo JSON' },
    33→  REQUEST: { code: -32600, message: 'Request JSON-RPC inválido' },
    34→  METHOD: { code: -32601, message: 'Método no encontrado' },
    35→  PARAMS: { code: -32602, message: 'Parámetros inválidos' },
    36→  INTERNAL: { code: -32603, message: 'Error interno' },
    37→}
    38→
    39→function rpcResult(id: JsonRpcMessage['id'], result: Record<string, unknown>) {
    40→  return { jsonrpc: '2.0', id, result }
    41→}
    42→
    43→function rpcError(id: JsonRpcMessage['id'], err: { code: number; message: string }, data?: unknown) {
    44→  return { jsonrpc: '2.0', id, error: { ...err, ...(data !== undefined ? { data } : {}) } }
    45→}

    70→function buildBundle(company: CompanyRow) {
    71→  return {
    72→    company: {
    73→      id: company.id,
    74→      name: company.name,
    75→      nit: company.nit,
    76→      description: company.description,
    77→      city: company.city,
    78→      department: company.department,
    79→      minBudget: company.minBudget,
    80→      maxBudget: company.maxBudget,
    81→      departmentsAllowed: parseJsonArray(company.departmentsAllowed),
    82→      modalitiesAllowed: parseJsonArray(company.modalitiesAllowed),
    83→      contractTypesAllowed: parseJsonArray(company.contractTypesAllowed),
    84→      phasesAllowed: parseJsonArray(company.phasesAllowed),
    85→      requireKeywordHit: company.requireKeywordHit,
    86→      capacity: company.capacity,
    87→      approverName: company.approverName,
    88→    } as CompanyConfigData,
    89→    products: company.products.map((p) => ({ id: p.id, name: p.name, description: p.description, category: p.category, keywords: parseJsonArray(p.keywords) })) as ProductItemData[],
    90→    experiences: company.experiences.map((e) => ({ id: e.id, title: e.title, entity: e.entity, year: e.year, value: e.value, description: e.description })) as ExperienceData[],
    91→    documents: company.documents.map((d) => ({ id: d.id, name: d.name, docType: d.docType, status: d.status, notes: d.notes })) as DocumentData[],
    92→  }
    93→}

    95→function mapRequirements(reqs: { code: string; description: string; status: string; evidence: string | null; action: string | null; source: string }[]) {
    96→  return reqs.map((r) => ({
    97→    code: r.code,
    98→    description: r.description,
    99→    status: r.status as 'CUMPLE' | 'NO_CUMPLE' | 'PENDIENTE' | 'REQUIERE_REVISION',
   100→    evidence: r.evidence ?? undefined,
   101→    action: r.action ?? undefined,
   102→    source: r.source as 'IA' | 'REGLA',
   103→  }))
   104→}

   107→  return {
   108→    id: p.id,
   109→    entidad: p.entity,
   110→    objeto: p.objectName,
   111→    descripcion: p.description,
   112→    valor_base_cop: p.basePrice,
   113→    modalidad: p.modality,
   114→    tipo_contrato: p.contractType,
   115→    fase: p.phase,
   116→    estado: p.state,
   117→    duracion: p.duration ? `${p.duration} ${p.durationUnit || ''}`.trim() : null,
   118→    ubicacion: [p.city, p.department].filter(Boolean).join(' / ') || null,
   119→    fecha_limite_aprox: p.receptionDate,
   120→    url: p.url,
   121→  }
   122→}

   140→function str(v: unknown, fallback = ''): string {

   143→function num(v: unknown): number | undefined {

   514→// ─── Dispatcher JSON-RPC ─────────────────────────────────────
   515→
   516→export interface RpcOutcome {
   517→  status: number
   518→  body: JsonRpcMessage | JsonRpcMessage[] | null
   519→  sessionId?: string
   520→}
   521→
   522→export async function handleRpcMessage(msg: JsonRpcMessage): Promise<RpcOutcome> {
   523→  if (!msg || typeof msg !== 'object' || msg.jsonrpc !== '2.0' || typeof msg.method !== 'string') {
   524→    return { status: 400, body: rpcError(msg?.id ?? null, ERR.REQUEST) }
   525→  }
   526→
   527→  const isNotification = msg.id === undefined || msg.id === null
   528→
   529→  switch (msg.method) {
   530→    case 'initialize': {
   531→      const requested = typeof msg.params?.protocolVersion === 'string' ? (msg.params.protocolVersion as string) : MCP_PROTOCOL_VERSION
   532→      const version = requested.startsWith('2025-') || requested.startsWith('2024-') ? requested : MCP_PROTOCOL_VERSION
   533→      return {
   534→        status: 200,
   535→        sessionId: crypto.randomUUID(),
   536→        body: rpcResult(msg.id, {
   537→          protocolVersion: version,
   538→          capabilities: { tools: { listChanged: false } },
   539→          serverInfo: MCP_SERVER_INFO,
   540→          instructions:
   541→            'Agente SECOP Radar: monitorea oportunidades de contratación pública colombiana (SECOP II), las analiza contra el perfil de la empresa y diseña proyectos-respuesta completos. Empieza con get_dashboard_stats o list_opportunities. Las tools de escritura respetan el flujo de aprobación humana.',
   542→        }),
   543→      }
   544→    }
   545→
   546→    case 'notifications/initialized':
   547→      return { status: 202, body: null }
   548→
   549→    case 'ping':
   550→      return isNotification ? { status: 202, body: null } : { status: 200, body: rpcResult(msg.id, {}) }
   551→
   552→    case 'tools/list':
   553→      return {
   554→        status: 200,
   555→        body: rpcResult(msg.id, {
   556→          tools: Object.entries(MCP_TOOLS).map(([name, t]) => ({
   557→            name,
   558→            title: name,
   559→            description: t.description + (t.write ? ' [escribe datos / exige aprobación humana previa]' : ''),
   560→            inputSchema: t.inputSchema,
   561→          })),
   562→        }),
   563→      }
   564→  }
   565→
   566→  if (msg.method.startsWith('notifications/')) return { status: 202, body: null }
   567→
   568→  if (msg.method === 'tools/call') {
   569→    const name = str(msg.params?.name)
   570→    const tool = MCP_TOOLS[name]
   571→    if (!tool) {
   572→      return { status: 200, body: rpcError(msg.id, ERR.METHOD, `Tool desconocida: ${name}. Usa tools/list para ver las disponibles.`) }
   573→    }
   574→    const args = (msg.params?.arguments as Record<string, unknown>) || {}
   575→    try {
   576→      const result = await tool.handler(args)
   577→      return {
   578→        status: 200,
   579→        body: rpcResult(msg.id, {
   580→          content: [{ type: 'text', text: JSON.stringify(result, null, 2) }],
   581→          structuredContent: result,
   582→          isError: false,
   583→        }),
   584→      }
   585→    } catch (err) {
   586→      const message = err instanceof Error ? err.message : 'Error interno de la tool'
   587→      return {
   588→        status: 200,
   589→        body: rpcResult(msg.id, { content: [{ type: 'text', text: `Error: ${message}` }], isError: true }),
   590→      }
   591→    }
   592→  }
   593→
   594→  return { status: 200, body: rpcError(msg.id, ERR.METHOD, `Método no soportado: ${msg.method}`) }
   595→}
