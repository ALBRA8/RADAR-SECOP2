    31→    31→function extractJson(raw: string): Record<string, unknown> | null {
    32→    32→  if (!raw) return null
    33→    33→  let text = raw.trim()
    34→    34→  const fence = text.match(/```(?:json)?\s*([\s\S]*?)```/i)
    35→    35→  if (fence) text = fence[1].trim()
    36→    36→  const start = text.indexOf('{')
    37→    37→  const end = text.lastIndexOf('}')
    38→    38→  if (start === -1 || end === -1 || end <= start) return null
    39→    39→  try {
    40→    40→    return JSON.parse(text.slice(start, end + 1))
    41→    41→  } catch {
    42→    42→    return null
    43→    43→  }

   325→   325→          content: String(s.content || ''),
   326→   326→          unconfirmed: Boolean(s.unconfirmed) || String(s.content || '').includes('[POR CONFIRMAR'),
   327→   327→        }))
   328→   328→        const checklists: ChecklistGroup[] = Array.isArray(parsed.checklists)
   329→   329→          ? (parsed.checklists as Record<string, unknown>[]).map((g) => ({
   330→   330→              group: String(g.group || 'Checklist'),
   331→   331→              items: Array.isArray(g.items)
   332→   332→                ? (g.items as Record<string, unknown>[]).map((it) => ({
   333→   333→                    label: String(it.label || ''),
   334→   334→                    required: Boolean(it.required),
   335→   335→                  }))
   336→   336→                : [],
   337→   337→            }))
   338→   338→          : []
   339→   339→        const unconfirmedCount =
   340→   340→          sections.filter((s) => s.unconfirmed).length +
   341→   341→          sections.reduce((acc, s) => acc + (String(s.content).match(/\[POR CONFIRMAR/g) || []).length, 0)
   342→   342→        return { sections, checklists, unconfirmedCount, engine: 'IA' }
   343→   343→      }
   344→   344→    } catch (err) {
   345→   345→      console.error('[ai] generateProposal: motor IA falló, cae a plantilla —', err instanceof Error ? err.message : err)
   346→   346→      // cae a plantilla
   347→   347→    }
   348→   348→  }
   349→   349→  return templateProposal(rec, bundle, requirements)
   350→   350→}
