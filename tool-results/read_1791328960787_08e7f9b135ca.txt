    46→export function extractJson(raw: string): Record<string, unknown> | null {
    47→  if (!raw) return null
    48→  let text = raw.trim()
    49→  const fence = text.match(/```(?:json)?\s*([\s\S]*?)```/i)
    50→  if (fence && fence[1].includes('{')) text = fence[1].trim()
    51→  const start = text.indexOf('{')
    52→  const end = text.lastIndexOf('}')
    53→  if (start === -1 || end === -1 || end <= start) return null
    54→  const body = text.slice(start, end + 1)
    55→  try {
    56→    return JSON.parse(body)
    57→  } catch {
    58→    // Reparación tolerante: los modelos suelen emitir saltos de línea literales
    59→    // dentro de los valores string (JSON inválido estricto). Estado in/out de string.
    60→    try {
    61→      return JSON.parse(repairControlChars(body))
    62→    } catch {
    63→      return null
    64→    }
    65→  }
    66→}
    67→
    68→/** Escapa caracteres de control (saltos, tabs) que estén DENTRO de strings JSON. */
    69→function repairControlChars(s: string): string {
    70→  let out = ''
    71→  let inString = false
    72→  for (let i = 0; i < s.length; i++) {
    73→    const ch = s[i]
    74→    if (inString) {
    75→      if (ch === '\\') {
    76→        out += ch + (s[i + 1] ?? '')
    77→        i++
    78→        continue
    79→      }
    80→      if (ch === '"') {
    81→        inString = false
    82→        out += ch
    83→        continue
    84→      }
    85→      const code = ch.charCodeAt(0)
    86→      if (code < 0x20) {
    87→        out += code === 10 ? '\\n' : code === 13 ? '\\r' : code === 9 ? '\\t' : `\\u${code.toString(16).padStart(4, '0')}`
    88→        continue
    89→      }
    90→      out += ch
    91→    } else {
    92→      if (ch === '"') inString = true
    93→      out += ch
    94→    }
    95→  }
    96→  return out
    97→}

   285→/** Normaliza y sanea el Marco Lógico que devuelve la IA (tolerante a faltantes). */
   286→export function normalizeMarcoLogico(raw: unknown): MarcoLogico | null {
   287→  if (!raw || typeof raw !== 'object') return null
   288→  const m = raw as Record<string, unknown>
   289→  const strArr = (v: unknown): string[] =>
   290→    Array.isArray(v) ? v.map((x) => String(x ?? '').trim()).filter(Boolean).slice(0, 8) : []
   291→  const filas: MarcoLogicoFila[] = (Array.isArray(m.filas) ? (m.filas as Record<string, unknown>[]) : [])
   292→    .filter((f) => f && typeof f === 'object')
   293→    .map((f) => ({
   294→      nivel: String(f.nivel ?? '').trim().slice(0, 80),
   295→      resumen: String(f.resumen ?? '').trim().slice(0, 700),
   296→      indicadores: String(f.indicadores ?? '').trim().slice(0, 700),
   297→      mediosVerificacion: String(f.mediosVerificacion ?? '').trim().slice(0, 450),
   298→      supuestos: String(f.supuestos ?? '').trim().slice(0, 450),
   299→    }))
   300→    .filter((f) => f.nivel && (f.resumen || f.indicadores))
   301→    .slice(0, 12)
   302→  if (filas.length === 0) return null
   303→  return {
   304→    problemaCentral: String(m.problemaCentral ?? '').trim().slice(0, 700),
   305→    causas: strArr(m.causas),
   306→    efectos: strArr(m.efectos),
   307→    objetivoCentral: String(m.objetivoCentral ?? '').trim().slice(0, 700),
   308→    filas,
   309→  }
   310→}
