    23→export interface JsonRpcMessage {
    24→  jsonrpc?: string
    25→  id?: string | number | null
    26→  method?: string
    27→  params?: Record<string, unknown>
    28→  result?: unknown
    29→  error?: { code: number; message: string; data?: unknown }
    30→}
    31→
    32→const ERR = {
    33→  PARSE: { code: -32700, message: 'Error de parseo JSON' },
    34→  REQUEST: { code: -32600, message: 'Request JSON-RPC inválido' },
    35→  METHOD: { code: -32601, message: 'Método no encontrado' },
    36→  PARAMS: { code: -32602, message: 'Parámetros inválidos' },
    37→  INTERNAL: { code: -32603, message: 'Error interno' },
    38→}
    39→
    40→function rpcResult(id: JsonRpcMessage['id'], result: Record<string, unknown>) {
    41→  return { jsonrpc: '2.0', id, result }
    42→}
    43→
    44→function rpcError(id: JsonRpcMessage['id'], err: { code: number; message: string }, data?: unknown) {
    45→  return { jsonrpc: '2.0', id, error: { ...err, ...(data !== undefined ? { data } : {}) } }
    46→}

    88→      approverName: company.approverName,
    89→    } as CompanyConfigData,
    90→    products: company.products.map((p) => ({ id: p.id, name: p.name, description: p.description, category: p.category, keywords: parseJsonArray(p.keywords) })) as ProductItemData[],

    96→function mapRequirements(reqs: { code: string; description: string; status: string; evidence: string | null; action: string | null; source: string }[]) {
    97→  return reqs.map((r) => ({
    98→    code: r.code,
    99→    description: r.description,
   100→    status: r.status as 'CUMPLE' | 'NO_CUMPLE' | 'PENDIENTE' | 'REQUIERE_REVISION',
   101→    evidence: r.evidence ?? undefined,
   102→    action: r.action ?? undefined,
   103→    source: r.source as 'IA' | 'REGLA',
   104→  }))
   105→}

   598→// ─── Dispatcher JSON-RPC ─────────────────────────────────────
   599→
   600→export interface RpcOutcome {
   601→  status: number
   602→  body: JsonRpcMessage | JsonRpcMessage[] | null
   603→  sessionId?: string
   604→}
   605→
   606→export async function handleRpcMessage(msg: JsonRpcMessage): Promise<RpcOutcome> {
   607→  if (!msg || typeof msg !== 'object' || msg.jsonrpc !== '2.0' || typeof msg.method !== 'string') {
   608→    return { status: 400, body: rpcError(msg?.id ?? null, ERR.REQUEST) }
   609→  }
   610→
   611→  const isNotification = msg.id === undefined || msg.id === null
   612→
   613→  switch (msg.method) {
   614→    case 'initialize': {
   615→      const requested = typeof msg.params?.protocolVersion === 'string' ? (msg.params.protocolVersion as string) : MCP_PROTOCOL_VERSION
   616→      const version = requested.startsWith('2025-') || requested.startsWith('2024-') ? requested : MCP_PROTOCOL_VERSION
   617→      return {
   618→        status: 200,
   619→        sessionId: crypto.randomUUID(),
   620→        body: rpcResult(msg.id, {
   621→          protocolVersion: version,
   622→          capabilities: { tools: { listChanged: false } },
   623→          serverInfo: MCP_SERVER_INFO,
   624→          instructions:
   625→            'Agente SECOP Radar: monitorea oportunidades de contratación pública colombiana (SECOP II), las analiza contra el perfil de la empresa, diseña proyectos-respuesta completos con Marco Lógico (árbol de problemas + matriz Fin/Propósito/Componentes/Actividades) y los refina conversacionalmente. También analiza pliegos/documentos y consulta la bandeja de canales de mensajería (Telegram/WhatsApp). Empieza con get_dashboard_stats o list_opportunities. Las tools de escritura respetan el flujo de aprobación humana.',
   626→        }),
   627→      }
   628→    }
   629→
   630→    case 'notifications/initialized':
   631→      return { status: 202, body: null }
   632→
   633→    case 'ping':
   634→      return isNotification ? { status: 202, body: null } : { status: 200, body: rpcResult(msg.id, {}) }
   635→
   636→    case 'tools/list':
   637→      return {
   638→        status: 200,
   639→        body: rpcResult(msg.id, {
   640→          tools: Object.entries(MCP_TOOLS).map(([name, t]) => ({
   641→            name,
   642→            title: name,
   643→            description: t.description + (t.write ? ' [escribe datos / exige aprobación humana previa]' : ''),
   644→            inputSchema: t.inputSchema,
   645→          })),
   646→        }),
   647→      }
   648→  }
   649→
   650→  if (msg.method.startsWith('notifications/')) return { status: 202, body: null }
   651→
   652→  if (msg.method === 'tools/call') {
   653→    const name = str(msg.params?.name)
   654→    const tool = MCP_TOOLS[name]
   655→    if (!tool) {
   656→      return { status: 200, body: rpcError(msg.id, ERR.METHOD, `Tool desconocida: ${name}. Usa tools/list para ver las disponibles.`) }
   657→    }
   658→    const args = (msg.params?.arguments as Record<string, unknown>) || {}
   659→    try {
   660→      const result = await tool.handler(args)
   661→      return {
   662→        status: 200,
   663→        body: rpcResult(msg.id, {
   664→          content: [{ type: 'text', text: JSON.stringify(result, null, 2) }],
   665→          structuredContent: result,
   666→          isError: false,
   667→        }),
   668→      }
   669→    } catch (err) {
   670→      const message = err instanceof Error ? err.message : 'Error interno de la tool'
   671→      return {
   672→        status: 200,
   673→        body: rpcResult(msg.id, { content: [{ type: 'text', text: `Error: ${message}` }], isError: true }),
   674→      }
   675→    }
   676→  }
   677→
   678→  return { status: 200, body: rpcError(msg.id, ERR.METHOD, `Método no soportado: ${msg.method}`) }
   679→}
