    42→  onBack: () => void
    43→  onAnalyze: (id: string) => void
    44→  onDiscard: (id: string, reason: string) => void

    46→  onGenerateProposal: (opportunityId: string) => void

    97→        </Button>

    99→          <CardContent className="p-5">
   100→            <div className="flex flex-col lg:flex-row gap-4">
   101→              <div className="flex gap-3 items-start shrink-0">
   102→                <ScoreBadge score={o.score} size="lg" />
   103→              </div>
   104→              <div className="flex-1 min-w-0">
   105→                <div className="flex flex-wrap items-center gap-2">
   106→                  <span className="font-mono text-xs bg-muted rounded px-1.5 py-0.5">{p.id}</span>
   107→                  <StatusBadge status={o.status} />
   108→                  {p.reference && <span className="text-xs text-muted-foreground">Ref.: {p.reference}</span>}
   109→                </div>
   110→                <h2 className="text-lg font-semibold mt-1.5 leading-snug">{p.objectName}</h2>
   111→                <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-x-4 gap-y-2 mt-3 text-sm">
   112→                  <InfoItem icon={Landmark} label="Entidad" value={p.entity} />
   113→                  <InfoItem icon={Building2} label="Ubicación" value={`${p.city || '—'} / ${p.department || '—'}`} />
   114→                  <InfoItem icon={Tag} label="Valor base" value={fmtCOP(p.basePrice)} strong />
   115→                  <InfoItem icon={Calendar} label="Publicación" value={fmtDate(p.publishDate)} />
   116→                  <InfoItem icon={Clock} label="Duración" value={p.duration ? `${p.duration} ${p.durationUnit || ''}` : 'No publicada'} />
   117→                  <InfoItem icon={FileCheck2} label="Modalidad" value={p.modality || '—'} />
   118→                  <InfoItem icon={FileCheck2} label="Tipo de contrato" value={p.contractType || '—'} />
   119→                  <InfoItem
   120→                    icon={ShieldAlert}
   121→                    label="Recepción de respuestas"
   122→                    value={
   123→                      p.receptionDate
   124→                        ? `${fmtDate(p.receptionDate)}${dDeadline != null ? ` (${dDeadline} día(s))` : ''}`

   127→                    strong={dDeadline != null && dDeadline <= 7}

   129→                </div>
   130→                {p.description && (
   131→                  <div className="mt-3">
   132→                    <Separator className="mb-3" />

   134→                  </div>

   136→                <div className="flex flex-wrap gap-2 mt-4">
   137→                  {p.url && (

   139→                      <Button size="sm" variant="outline" className="min-h-9">
   140→                        <ExternalLink className="w-4 h-4 mr-1.5" aria-hidden /> Ver en SECOP
   141→                      </Button>
   142→                    </a>
   143→                  )}
   144→                  {active && (o.status === 'NUEVA' || o.status === 'COMPATIBLE' || o.status === 'EN_ANALISIS') && (
   145→                    <Button size="sm" onClick={() => onAnalyze(o.id)} disabled={analyzing} className="bg-emerald-600 hover:bg-emerald-700 min-h-9">
   146→                      <FilePlus2 className="w-4 h-4 mr-1.5" aria-hidden />
   147→                      {analyzing ? 'Analizando con IA…' : o.requirements && o.requirements.length > 0 ? 'Re-analizar con IA' : 'Analizar con IA'}
   148→                    </Button>
   149→                  )}
   150→                  {active && o.status === 'COMPATIBLE' && (
   151→                    <Button size="sm" variant="outline" onClick={() => onApprovePrep(o.id)} className="border-lime-500 text-lime-700 hover:bg-lime-50 min-h-9">
   152→                      <CheckCircle2 className="w-4 h-4 mr-1.5" aria-hidden /> Aprobar para preparación
   153→                    </Button>
   154→                  )}
   155→                  {active && o.status === 'APROBADA_PREPARACION' && (
   156→                    <Button size="sm" onClick={() => onGenerateProposal(o.id)} disabled={generating} className="bg-lime-600 hover:bg-lime-700 min-h-9">
   157→                      <FileText className={`w-4 h-4 mr-1.5 ${generating ? 'animate-pulse' : ''}`} aria-hidden />
   158→                      {generating ? 'Generando borrador…' : latestProposal ? 'Generar nueva versión' : 'Generar borrador de propuesta'}
   159→                    </Button>
   160→                  )}
   161→                  {active && (
   162→                    <Button size="sm" variant="outline" onClick={() => setDiscardOpen(true)} className="border-rose-300 text-rose-700 hover:bg-rose-50 min-h-9">
   163→                      <Ban className="w-4 h-4 mr-1.5" aria-hidden /> Descartar
   164→                    </Button>
   165→                  )}
   166→                </div>
   167→                {o.status === 'DESCARTADA' && o.discardReason && (
   168→                  <p className="text-sm text-rose-700 bg-rose-50 border border-rose-200 rounded px-3 py-2 mt-3">
   169→                    Descartada: {o.discardReason}
   170→                  </p>
   171→                )}
   172→              </div>
   173→            </div>
   174→          </CardContent>
   175→        </Card>
   176→      </div>
   177→
   178→      <Tabs defaultValue="compatibilidad">
   179→        <TabsList className="w-full sm:w-auto flex-wrap h-auto">
   180→          <TabsTrigger value="compatibilidad">Compatibilidad explicada</TabsTrigger>
   181→          <TabsTrigger value="matriz">Matriz de requisitos</TabsTrigger>
   182→          <TabsTrigger value="riesgos">Riesgos y faltantes</TabsTrigger>
   183→          <TabsTrigger value="propuesta">Propuesta {latestProposal ? `· v${latestProposal.version}` : ''}</TabsTrigger>
   184→        </TabsList>
   185→
   186→        {/* Compatibilidad */}
   187→        <TabsContent value="compatibilidad" className="mt-4 space-y-4">
   188→          {o.analysisSummary && (
   189→            <Card>
   190→              <CardContent className="p-4">
   191→                <p className="text-sm leading-relaxed">{o.analysisSummary}</p>
   192→              </CardContent>
   193→            </Card>
   194→          )}
   195→          <Card>
   196→            <CardHeader className="pb-2">
   197→              <CardTitle className="text-base">¿Por qué esta oportunidad es (o no) compatible?</CardTitle>
   198→              <p className="text-sm text-muted-foreground">Cada dimensión se evalúa contra el perfil registrado de la empresa. El sistema muestra las razones, no solo el porcentaje.</p>
   199→            </CardHeader>
   200→            <CardContent className="space-y-3">
   201→              {o.reasons.length === 0 && <p className="text-sm text-muted-foreground">Ejecuta una sincronización para recalcular la compatibilidad.</p>}
   202→              {o.reasons.map((d) => (
   203→                <div key={d.key} className="flex items-start gap-3 rounded-lg border p-3">
   204→                  <DimIcon result={d.result} />
   205→                  <div className="flex-1 min-w-0">
   206→                    <div className="flex flex-wrap items-center justify-between gap-2">
   207→                      <p className="font-medium text-sm">{d.label}</p>
   208→                      <Badge variant="secondary" className="text-[11px]">
   209→                        peso {Math.round(d.weight * 100)}% · {Math.round(d.score * 100)}%
   210→                      </Badge>
   211→                    </div>
   212→                    <p className={`text-sm mt-0.5 ${DIM_COLORS[d.result]}`}>{d.reason}</p>
   213→                  </div>
   214→                </div>
   215→              ))}
   216→            </CardContent>
   217→          </Card>
   218→        </TabsContent>
   219→
   220→        {/* Matriz de requisitos */}
   221→        <TabsContent value="matriz" className="mt-4">
   222→          <Card>
   223→            <CardHeader className="pb-2">
   224→              <CardTitle className="text-base">Matriz de cumplimiento</CardTitle>
   225→              <p className="text-sm text-muted-foreground">
   226→                REQUISITO | ESTADO | EVIDENCIA | ACCIÓN — construida con la evidencia registrada de la empresa. Nada se marca «Cumple» sin sustento.
   227→              </p>
   228→            </CardHeader>
   229→            <CardContent>
   230→              {!o.requirements || o.requirements.length === 0 ? (
   231→                <p className="text-sm text-muted-foreground py-4">
   232→                  Aún no hay matriz. Usa «Analizar con IA» para generarla.
   233→                </p>
   234→              ) : (
   235→                <div className="overflow-x-auto">
   236→                  <table className="w-full text-sm border-collapse">
   237→                    <thead>
   238→                      <tr className="border-b text-left text-xs text-muted-foreground uppercase tracking-wide">

   246→                      {o.requirements.map((r) => (

   251→                            <Badge variant="secondary" className="ml-2 text-[10px]">{r.source === 'IA' ? 'IA' : 'Reglas'}</Badge>

   253→                          <td className="py-3 pr-3"><ReqStatusBadge status={r.status} /></td>
   254→                          <td className="py-3 pr-3 text-muted-foreground">{r.evidence || '—'}</td>
   255→                          <td className="py-3 text-muted-foreground">{r.action || '—'}</td>
   256→                        </tr>
   257→                      ))}
   258→                    </tbody>
   259→                  </table>
   260→                </div>
   261→              )}
   262→            </CardContent>
   263→          </Card>
   264→        </TabsContent>
   265→
   266→        {/* Riesgos */}
   267→        <TabsContent value="riesgos" className="mt-4 grid md:grid-cols-2 gap-4">
   268→          <Card>
   269→            <CardHeader className="pb-2">
   270→              <CardTitle className="text-base flex items-center gap-2"><ShieldAlert className="w-4 h-4 text-rose-600" aria-hidden /> Riesgos identificados</CardTitle>
   271→            </CardHeader>
   272→            <CardContent>
   273→              {o.risks.length === 0 ? (
   274→                <p className="text-sm text-muted-foreground">Sin riesgos registrados en el último análisis.</p>
   275→              ) : (
   276→                <ul className="space-y-2">
   277→                  {o.risks.map((r, i) => (
   278→                    <li key={i} className="text-sm rounded-md border border-rose-200 bg-rose-50 px-3 py-2 text-rose-900">{r}</li>
   279→                  ))}
   280→                </ul>
   281→              )}
   282→            </CardContent>
   283→          </Card>
   284→          <Card>
   285→            <CardHeader className="pb-2">
   286→              <CardTitle className="text-base flex items-center gap-2"><FileCheck2 className="w-4 h-4 text-amber-600" aria-hidden /> Documentos faltantes</CardTitle>
   287→            </CardHeader>
   288→            <CardContent>
   289→              {o.missingDocs.length === 0 ? (
   290→                <p className="text-sm text-muted-foreground">No se detectaron documentos faltantes en el expediente de la empresa.</p>
   291→              ) : (
   292→                <ul className="space-y-2">
   293→                  {o.missingDocs.map((d, i) => (
   294→                    <li key={i} className="text-sm rounded-md border border-amber-200 bg-amber-50 px-3 py-2 text-amber-900">{d}</li>
   295→                  ))}
   296→                </ul>
   297→              )}
   298→              {o.nextAction && (
   299→                <>
   300→                  <Separator className="my-3" />
   301→                  <p className="text-sm"><strong>Siguiente acción:</strong> {o.nextAction}</p>
   302→                </>
   303→              )}
   304→            </CardContent>
   305→          </Card>
   306→        </TabsContent>
   307→
   308→        {/* Propuesta */}
   309→        <TabsContent value="propuesta" className="mt-4 space-y-4">
   310→          {o.status !== 'APROBADA_PREPARACION' ? (
   311→            <Card>
   312→              <CardContent className="p-6 text-center">
   313→                <FileText className="w-8 h-8 mx-auto text-muted-foreground" aria-hidden />
   314→                <p className="mt-2 font-medium">La propuesta se genera después de la aprobación humana</p>
   315→                <p className="text-sm text-muted-foreground mt-1 max-w-lg mx-auto">
   316→                  Cuando la oportunidad esté analizada y quieras prepararla, apruébala para preparación (Módulo H). Así queda trazabilidad de quién autorizó cada paso.
   317→                </p>
   318→              </CardContent>
   319→            </Card>
   320→          ) : !latestProposal ? (
   321→            <Card>
   322→              <CardContent className="p-6 text-center">
   323→                <FileText className="w-8 h-8 mx-auto text-muted-foreground" aria-hidden />
   324→                <p className="mt-2 font-medium">Oportunidad aprobada — lista para preparar</p>
   325→                <p className="text-sm text-muted-foreground mt-1">Genera el borrador: el agente usará únicamente la información registrada de la empresa y marcará lo pendiente.</p>
   326→                <Button onClick={() => onGenerateProposal(o.id)} disabled={generating} className="mt-4 bg-lime-600 hover:bg-lime-700 min-h-11">
   327→                  <FileText className="w-4 h-4 mr-2" aria-hidden /> {generating ? 'Generando…' : 'Generar borrador'}
   328→                </Button>
   329→              </CardContent>
   330→            </Card>
   331→          ) : (
   332→            <ProposalPanel
   333→              proposal={latestProposal}
   334→              messages={o.messages || []}
   335→              generating={generating}
   336→              refining={refining}
   337→              onGenerate={() => onGenerateProposal(o.id)}
   338→              onApprove={() => setApproveOpen(latestProposal)}
   339→              onReject={(notes) => onApproveProposal(latestProposal.id, 'RECHAZADA', approver || 'Responsable', notes)}
   340→              onRefine={(instruction) => onRefineProposal(latestProposal.id, instruction)}
   341→            />
   342→          )}
   343→        </TabsContent>
   344→      </Tabs>
   345→
   346→      {/* Diálogo descarte */}
   347→      <Dialog open={discardOpen} onOpenChange={setDiscardOpen}>
   348→        <DialogContent>
   349→          <DialogHeader>
   350→            <DialogTitle>Descartar oportunidad</DialogTitle>
   351→          </DialogHeader>
   352→          <div className="space-y-2">
   353→            <Label htmlFor="motivo">Motivo del descarte (obligatorio, queda en auditoría)</Label>
   354→            <Textarea id="motivo" value={discardReason} onChange={(e) => setDiscardReason(e.target.value)} placeholder="Ej.: fuera de la cobertura geográfica del área operativa…" rows={3} />
   355→          </div>
   356→          <DialogFooter>
   357→            <Button variant="outline" onClick={() => setDiscardOpen(false)}>Cancelar</Button>
   358→            <Button variant="destructive" onClick={submitDiscard}>Descartar</Button>
   359→          </DialogFooter>
   360→        </DialogContent>
   361→      </Dialog>
   362→
   363→      {/* Diálogo aprobación */}
   364→      <Dialog open={!!approveOpen} onOpenChange={(open) => !open && setApproveOpen(null)}>
   365→        <DialogContent>
   366→          <DialogHeader>
   367→            <DialogTitle>Aprobación de la propuesta (Módulo H)</DialogTitle>
   368→          </DialogHeader>
   369→          <div className="space-y-3">
   370→            <p className="text-sm text-muted-foreground">
   371→              Al aprobar declaras que revisaste la matriz de requisitos, los documentos y el borrador. La decisión queda registrada con fecha y responsable.
   372→            </p>
   373→            <div className="space-y-1.5">
   374→              <Label htmlFor="aprobador">Nombre del responsable que aprueba</Label>
   375→              <Input id="aprobador" value={approver} onChange={(e) => setApprover(e.target.value)} placeholder="Ej.: María Gómez — Gerente" />
   376→            </div>
   377→            <div className="space-y-1.5">
   378→              <Label htmlFor="notas">Notas (opcional)</Label>
   379→              <Textarea id="notas" value={approverNotes} onChange={(e) => setApproverNotes(e.target.value)} rows={2} placeholder="Condiciones confirmadas, precio aprobado, etc." />
   380→            </div>
   381→          </div>
   382→          <DialogFooter className="sm:justify-between">
   383→            <Button
   384→              variant="outline"
   385→              className="border-rose-300 text-rose-700 hover:bg-rose-50"
   386→              onClick={() => {
   387→                if (!approveOpen) return
   388→                onApproveProposal(approveOpen.id, 'RECHAZADA', approver.trim() || 'Responsable', approverNotes.trim() || 'Devuelta para ajustes')
   389→                setApproveOpen(null)
   390→                setApprover('')
   391→                setApproverNotes('')
   392→              }}
   393→            >
   394→              <Ban className="w-4 h-4 mr-1.5" aria-hidden /> Devolver para ajustes
   395→            </Button>
   396→            <div className="flex gap-2">
   397→              <Button variant="outline" onClick={() => setApproveOpen(null)}>Cancelar</Button>
   398→              <Button onClick={submitApproval} className="bg-emerald-600 hover:bg-emerald-700">
   399→                <UserCheck className="w-4 h-4 mr-1.5" aria-hidden /> Aprobar propuesta
   400→              </Button>
   401→            </div>
   402→          </DialogFooter>
   403→        </DialogContent>
   404→      </Dialog>
   405→    </div>
   406→  )
   407→}

   409→function InfoItem({ icon: Icon, label, value, strong }: { icon: typeof Landmark; label: string; value: string; strong?: boolean }) {
   410→  return (
   411→    <div className="flex items-start gap-2 min-w-0">
   412→      <Icon className="w-4 h-4 mt-0.5 text-muted-foreground shrink-0" aria-hidden />
   413→      <div className="min-w-0">
   414→        <p className="text-xs text-muted-foreground">{label}</p>
   415→        <p className={`text-sm leading-snug ${strong ? 'font-semibold' : 'font-medium'} break-words`}>{value}</p>
   416→      </div>
   417→    </div>
   418→  )
   419→}
