    23→export interface JsonRpcMessage {
    24→  jsonrpc?: string
    25→  id?: string | number | null
    26→  method?: string
    27→  params?: Record<string, unknown>
    28→  result?: unknown
    29→  error?: { code: number; message: string; data?: unknown }
    30→}
    31→
    32→const ERR = {
    33→  PARSE: { code: -32700, message: 'Error de parseo JSON' },
    34→  REQUEST: { code: -32600, message: 'Request JSON-RPC inválido' },
    35→  METHOD: { code: -32601, message: 'Método no encontrado' },
    36→  PARAMS: { code: -32602, message: 'Parámetros inválidos' },
    37→  INTERNAL: { code: -32603, message: 'Error interno' },
    38→}
    39→
    40→function rpcResult(id: JsonRpcMessage['id'], result: Record<string, unknown>) {
    41→  return { jsonrpc: '2.0', id, result }
    42→}
    43→
    44→function rpcError(id: JsonRpcMessage['id'], err: { code: number; message: string }, data?: unknown) {
    45→  return { jsonrpc: '2.0', id, error: { ...err, ...(data !== undefined ? { data } : {}) } }
    46→}

    96→function mapRequirements(reqs: { code: string; description: string; status: string; evidence: string | null; action: string | null; source: string }[]) {
    97→  return reqs.map((r) => ({
    98→    code: r.code,
    99→    description: r.description,
   100→    status: r.status as 'CUMPLE' | 'NO_CUMPLE' | 'PENDIENTE' | 'REQUIERE_REVISION',
   101→    evidence: r.evidence ?? undefined,
   102→    action: r.action ?? undefined,
   103→    source: r.source as 'IA' | 'REGLA',
   104→  }))
   105→}

   606→// ─── Dispatcher JSON-RPC ─────────────────────────────────────
   607→
   608→export interface RpcOutcome {
   609→  status: number
   610→  body: JsonRpcMessage | JsonRpcMessage[] | null
   611→  sessionId?: string
   612→}
   613→
   614→export async function handleRpcMessage(msg: JsonRpcMessage): Promise<RpcOutcome> {
   615→  if (!msg || typeof msg !== 'object' || msg.jsonrpc !== '2.0' || typeof msg.method !== 'string') {
   616→    return { status: 400, body: rpcError(msg?.id ?? null, ERR.REQUEST) }
   617→  }
   618→
   619→  const isNotification = msg.id === undefined || msg.id === null
   620→
   621→  switch (msg.method) {
   622→    case 'initialize': {
   623→      const requested = typeof msg.params?.protocolVersion === 'string' ? (msg.params.protocolVersion as string) : MCP_PROTOCOL_VERSION
   624→      const version = requested.startsWith('2025-') || requested.startsWith('2024-') ? requested : MCP_PROTOCOL_VERSION
   625→      return {
   626→        status: 200,
   627→        sessionId: crypto.randomUUID(),
   628→        body: rpcResult(msg.id, {
   629→          protocolVersion: version,
   630→          capabilities: { tools: { listChanged: false } },
   631→          serverInfo: MCP_SERVER_INFO,
   632→          instructions:
   633→            'Agente SECOP Radar: monitorea oportunidades de contratación pública colombiana (SECOP II), las analiza contra el perfil de la empresa, diseña proyectos-respuesta completos con Marco Lógico (árbol de problemas + matriz Fin/Propósito/Componentes/Actividades) y los refina conversacionalmente. También analiza pliegos/documentos y consulta la bandeja de canales de mensajería (Telegram/WhatsApp). Empieza con get_dashboard_stats o list_opportunities. Las tools de escritura respetan el flujo de aprobación humana.',
   634→        }),
   635→      }
   636→    }
   637→
   638→    case 'notifications/initialized':
   639→      return { status: 202, body: null }
   640→
   641→    case 'ping':
   642→      return isNotification ? { status: 202, body: null } : { status: 200, body: rpcResult(msg.id, {}) }
   643→
   644→    case 'tools/list':
   645→      return {
   646→        status: 200,
   647→        body: rpcResult(msg.id, {
   648→          tools: Object.entries(MCP_TOOLS).map(([name, t]) => ({
   649→            name,
   650→            title: name,
   651→            description: t.description + (t.write ? ' [escribe datos / exige aprobación humana previa]' : ''),
   652→            inputSchema: t.inputSchema,
   653→          })),
   654→        }),
   655→      }
   656→  }
   657→
   658→  if (msg.method.startsWith('notifications/')) return { status: 202, body: null }
   659→
   660→  if (msg.method === 'tools/call') {
   661→    const name = str(msg.params?.name)
   662→    const tool = MCP_TOOLS[name]
   663→    if (!tool) {
   664→      return { status: 200, body: rpcError(msg.id, ERR.METHOD, `Tool desconocida: ${name}. Usa tools/list para ver las disponibles.`) }
   665→    }
   666→    const args = (msg.params?.arguments as Record<string, unknown>) || {}
   667→    try {
   668→      const result = await tool.handler(args)
   669→      return {
   670→        status: 200,
   671→        body: rpcResult(msg.id, {
   672→          content: [{ type: 'text', text: JSON.stringify(result, null, 2) }],
   673→          structuredContent: result,
   674→          isError: false,
   675→        }),
   676→      }
   677→    } catch (err) {
   678→      const message = err instanceof Error ? err.message : 'Error interno de la tool'
   679→      return {
   680→        status: 200,
   681→        body: rpcResult(msg.id, { content: [{ type: 'text', text: `Error: ${message}` }], isError: true }),
   682→      }
   683→    }
   684→  }
   685→
   686→  return { status: 200, body: rpcError(msg.id, ERR.METHOD, `Método no soportado: ${msg.method}`) }
   687→}
